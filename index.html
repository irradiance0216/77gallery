// netlify/functions/check-availability.mjs
// Native ES Module for Node.js 22 runtime

export async function handler(event, context) {
  // 1. In-Memory Header Auth Verification
  const authHeader = event.headers.authorization || '';
  const expectedToken = process.env.TRACKER_PASSCODE || 'oshikatsu2026';

  if (!authHeader.includes(expectedToken)) {
    return {
      statusCode: 401,
      body: JSON.stringify({ error: 'Unauthorized: Passcode invalid or missing' }),
    };
  }

  // 2. Parse Hotel Parameters
  const params = event.queryStringParameters || {};
  const hotelCode = params.chl || '00042'; // Default: Tokyo Haneda Airport No.1
  const checkinDate = params.ch_in || '2026-10-15';

  try {
    const targetUrl = `https://www.toyoko-inn.com/search/result?chl=${hotelCode}&ch_in=${checkinDate}`;
    
    // Native Node 22 fetch call
    const response = await fetch(targetUrl, {
      headers: {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
      }
    });

    const htmlText = await response.text();

    const hasAvailable = htmlText.includes('空室あり') || htmlText.includes('残りわずか');
    const isFull = htmlText.includes('満室');

    return {
      statusCode: 200,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        hotelCode,
        checkinDate,
        status: hasAvailable ? 'Available' : (isFull ? 'Full' : 'Unknown'),
        timestamp: new Date().toISOString(),
        officialUrl: targetUrl
      }),
    };
  } catch (error) {
    return {
      statusCode: 500,
      body: JSON.stringify({ error: 'Scraper failed', details: error.message }),
    };
  }
}
