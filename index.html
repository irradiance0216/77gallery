export async function handler(event, context) {
  const { hotel, start, end, room = 1, people = 1, smoking = 'noSmoking' } = event.queryStringParameters || {};

  if (!hotel || !start || !end) {
    return { statusCode: 400, body: JSON.stringify({ error: "Missing parameters" }) };
  }

  try {
    // Queries Toyoko Inn's internal JSON endpoint directly, avoiding HTML shell caches
    const apiUrl = `https://www.toyoko-inn.com/api/v1/search/room_plans?hotel=${hotel}&start=${start}&end=${end}&people=${people}&room=${room}&smoking=${smoking}`;

    const response = await fetch(apiUrl, {
      headers: {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) Chrome/128.0.0.0 Safari/537.36',
        'Accept': 'application/json, text/plain, */*',
        'Referer': `https://www.toyoko-inn.com/search/result/room_plan/?hotel=${hotel}&start=${start}&end=${end}`,
        'Cache-Control': 'no-cache, no-store, must-revalidate',
        'Pragma': 'no-cache'
      }
    });

    if (!response.ok) throw new Error(`API returned ${response.status}`);

    const data = await response.json();
    const availablePlans = data.room_plans || data.plans || [];
    const isAvailable = availablePlans.length > 0 && availablePlans.some(p => p.vacant_count > 0 || p.is_bookable);

    return {
      statusCode: 200,
      headers: { 
        'Content-Type': 'application/json',
        'Cache-Control': 'no-store, max-age=0' 
      },
      body: JSON.stringify({
        hotel, start, end,
        available: isAvailable,
        planCount: availablePlans.length,
        timestamp: new Date().toISOString()
      })
    };
  } catch (err) {
    return { statusCode: 500, body: JSON.stringify({ error: err.message, available: false }) };
  }
}
